    ```python
    import html
    from datetime import datetime
    from typing import Optional

    from sqlalchemy import func, desc

    from app.database import (
        SessionLocal, DigestEvent, IntruderEvent, BlockedIP, AccessLog,
    )
    from app.telegram_alerter import send_alert
    from app.geoip import lookup_batch

    SAFETY_LIMIT = 4000  # UTF-16 code-unit safety margin below Telegram's 4096 hard limit

    # Emoji + reason labels
    REASON_LABELS = {
        "sql_injection":    ("💉", "SQL injection"),
        "honeypot":         ("🕳️", "Honeypot hit"),
        "suspicious_path":  ("🔍", "Suspicious path scan"),
        "rate_limit":       ("🚦", "Rate limit"),
        "auth_brute_force": ("🔐", "Auth brute-force"),
    }
    DEFAULT_REASON_EMOJI = "⚠️"


    def _tg_len(s: str) -> int:
        """Length in UTF-16 code units — Telegram's 4096 measurement unit."""
        return len(s.encode("utf-16-le")) // 2


    def _esc(s: Optional[str]) -> str:
        """HTML-escape for insertion between HTML tags (not inside attributes)."""
        if s is None:
            return ""
        return html.escape(str(s), quote=False)

    async def _render(db, rows, intruder_ids, block_ids, min_ts, max_ts,
                     top_n: int, include_paths: bool, orphan_count: int) -> str:
        parts = [
            _build_header(min_ts, max_ts, len(rows)),
            _build_blocked_section(db, block_ids),
            _build_attack_breakdown(db, intruder_ids),
            await _build_top_attackers(db, intruder_ids, top_n=top_n, include_paths=include_paths),
            _build_traffic_overview(db, min_ts, max_ts),
            _build_footer(len(rows), orphan_count),
        ]
        return "\n".join(p for p in parts if p)


    async def _truncate_if_needed(db, rows, intruder_ids, block_ids, min_ts, max_ts, orphan_count) -> str:
        # Iterative rebuild per D-15: attacker list 10 -> 5 -> 3 (paths on), then 3 (paths off).
        # Max 4 renders; accept last if still over.
        attempts = [
            (10, True), (5, True), (3, True), (3, False),
        ]
        msg = ""
        for top_n, include_paths in attempts:
            msg = await _render(db, rows, intruder_ids, block_ids, min_ts, max_ts,
                                top_n=top_n, include_paths=include_paths,
                                orphan_count=orphan_count)
            if _tg_len(msg) <= SAFETY_LIMIT:
                return msg
        return msg  # accept last attempt even if over (D-15 "whatever fits after passes")


    async def build_message(db, rows) -> tuple[str, list[int]]:
        """Assemble the digest HTML message. Returns (message, included_row_ids).
        Pure: does not send, does not update sent_at. Shared by send_digest and preview."""
        intruder_ids = [r.source_id for r in rows if r.source == "intruder"]
        block_ids    = [r.source_id for r in rows if r.source == "auto_block"]

        # Orphan detection (per CONTEXT <specifics>): compare requested ids to hydrated rows.
        intruder_rows = (db.query(IntruderEvent).filter(IntruderEvent.id.in_(intruder_ids)).all()
                         if intruder_ids else [])
        block_rows = (db.query(BlockedIP).filter(BlockedIP.id.in_(block_ids)).all()
                      if block_ids else [])
        orphan_count = (len(intruder_ids) - len(intruder_rows)) + (len(block_ids) - len(block_rows))

        min_ts = min(r.timestamp for r in rows)
        max_ts = max(r.timestamp for r in rows)

        message = await _truncate_if_needed(
            db, rows, intruder_ids, block_ids, min_ts, max_ts, orphan_count
        )
        return message, [r.id for r in rows]


    async def send_digest() -> dict:
        """Entry point — manually triggered and Phase-3 scheduled.
        Returns {sent, event_count, skipped_reason, telegram_ok, message}."""
        db = SessionLocal()
        try:
            rows = db.query(DigestEvent).filter(DigestEvent.sent_at.is_(None)).all()
            if not rows:
                return {
                    "sent": False, "event_count": 0,
                    "skipped_reason": "no_events", "telegram_ok": False,
                    "message": None,
                }

            message, included_ids = await build_message(db, rows)

            telegram_ok = await send_alert(message, parse_mode="HTML")

            if telegram_ok:
                now = datetime.utcnow()
                (db.query(DigestEvent)
                   .filter(DigestEvent.id.in_(included_ids))
                   .update({DigestEvent.sent_at: now}, synchronize_session=False))
                db.commit()
                return {
                    "sent": True, "event_count": len(included_ids),
                    "skipped_reason": None, "telegram_ok": True,
                    "message": message,
                }

            print(f"Digest send failed: Telegram returned non-200; {len(included_ids)} rows stay unsent")
            return {
                "sent": False, "event_count": len(included_ids),
                "skipped_reason": "telegram_failed", "telegram_ok": False,
                "message": message,
            }
        finally:
            db.close()
    ```

| Boundary | Description |
|----------|-------------|
| attacker → access log → Traefik → intruder_events/blocked_ips | Attacker-controlled strings (ip, path, user_agent, host, country name) land in DB; digest reads and embeds them in an HTML-parse-mode Telegram message |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Tampering | `app/digest.py` message assembly — HTML injection via attacker-controlled IntruderEvent.path / host / user_agent / BlockedIP country | mitigate | `html.escape(value, quote=False)` on every attacker field (ip, path, host, user_agent, country) at interpolation time, per D-16; NEVER escape assembled strings; only structural `<b>`, `<code>`, `<i>` tags inserted raw. Verified by `test_html_escape` |
| T-02-02 | DoS | `app/digest.py` — oversized message triggers Telegram 400 "message too long" | mitigate | UTF-16 code-unit measurement (`len(s.encode("utf-16-le")) // 2`) with 4000-unit safety margin; iterative truncation loop per D-15. Verified by `test_truncation` |
| T-02-03 | DoS | Telegram parse error from malformed HTML entity in attacker input (e.g. raw `&` in path) | mitigate | `html.escape` handles `&` → `&amp;`, `<` → `&lt;`, `>` → `&gt;` — standard stdlib, verified semantics |
