    ```python
    """Alert routing: severity computation + immediate/digest gate + digest persistence.

    Leaf module in the alerting subgraph. Imports only from app.config and app.database.
    MUST NOT import from telegram_alerter, log_watcher, or auto_blocker (circular-import risk).
    """
    from datetime import datetime
    from typing import Literal
    from sqlalchemy.orm import Session
    from app.config import ALERT_MIN_SEVERITY
    from app.database import DigestEvent


    # Severity classification (moved from telegram_alerter.py per D-01)
    THREAT_SEVERITY: dict[str, str] = {
        "sql_injection": "critical",
        "honeypot": "critical",
        "auth_failures": "high",
        "rate_limit": "medium",
        "suspicious_path": "medium",
    }

    # Integer rank for severity comparison (D-03)
    _SEVERITY_RANK: dict[str, int] = {"critical": 3, "high": 2, "medium": 1}


    def get_severity(reason: str, event: dict) -> Literal["critical", "high", "medium"]:
        """Return severity for an intruder event, honoring request-count escalation.

        Preserves the rule previously implemented in telegram_alerter.get_severity_header (D-06):
        - request_count > 50 -> escalate to critical
        - request_count > 20 and base severity == medium -> escalate to high
        Unknown reasons fall back to "medium".
        """
        severity = THREAT_SEVERITY.get(reason, "medium")
        request_count = event.get("request_count", 1) or 1

        if request_count > 50:
            severity = "critical"
        elif request_count > 20 and severity == "medium":
            severity = "high"

        return severity  # type: ignore[return-value]


    def route_event(event: dict) -> Literal["immediate", "digest"]:
        """Decide whether an intruder event fires immediately or waits for the digest.

        Returns "immediate" when rank(severity) >= rank(ALERT_MIN_SEVERITY), else "digest".
        """
        severity = get_severity(event.get("reason", ""), event)
        threshold_rank = _SEVERITY_RANK.get(ALERT_MIN_SEVERITY, _SEVERITY_RANK["high"])
        event_rank = _SEVERITY_RANK.get(severity, _SEVERITY_RANK["medium"])
        return "immediate" if event_rank >= threshold_rank else "digest"


    def persist_to_digest(
        db: Session,
        *,
        source: str,
        source_id: int,
        severity: str,
    ) -> None:
        """Persist a digest-eligible event reference. Commits inline. Synchronous (D-11)."""
        entry = DigestEvent(
            timestamp=datetime.utcnow(),
            source=source,
            source_id=source_id,
            severity=severity,
            sent_at=None,
        )
        db.add(entry)
        db.commit()
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| attacker → intruder event dict | `event["reason"]`, `event["request_count"]`, `event["ip"]` originate from parsed access logs (attacker-controlled). They flow into `get_severity` and `route_event`. |
| watchdog thread → SQLite | `persist_to_digest` writes from the log watcher thread. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-06 | Tampering | `get_severity(reason, event)` | mitigate | `reason` is looked up in a fixed dict (`THREAT_SEVERITY`); unknown keys fall back to `"medium"`. `request_count` is compared numerically only; no eval/format. Attacker cannot inject a higher-or-lower severity by crafting a weird reason string — unknown reasons deterministically map to "medium". |
| T-01-07 | Elevation of Privilege | `route_event` bypass | mitigate | No code path makes routing depend on attacker-supplied severity string directly; severity is always computed from `get_severity`. `ALERT_MIN_SEVERITY` is validated in config.py (Plan 01). Belt-and-suspenders: unknown `ALERT_MIN_SEVERITY` inside `route_event` falls back to `"high"` rank. |
| T-01-08 | Tampering (SQL) | `persist_to_digest` SQL | mitigate | SQLAlchemy ORM `db.add(DigestEvent(...))` — fully parameterized. Only callers pass `source_id: int` (PK reference) and enum-valued `source`/`severity` strings. Attacker IP/path not stored. |
| T-01-09 | Denial of Service | `persist_to_digest` commit-per-call | accept | D-11 explicitly forbids batching in Phase 1. Commit-per-event is consistent with existing `log_watcher.process_line` commit pattern (lines 116, 131) and `auto_blocker` session handling. Burst risk mitigated by SQLite's WAL (default sessionmaker). |
| T-01-10 | Information Disclosure | circular import leak | mitigate | `alert_router.py` imports only from `app.config` + `app.database` (leaf rule). Grep-enforced in acceptance criteria. Prevents accidental pull of Telegram token into router namespace. |
| T-01-11 | Spoofing | forged `source`/`source_id` | accept | `persist_to_digest` trusts its callers (only `log_watcher` and `auto_blocker` per D-02/D-05 — enforced by Plan 03). No external input path to this function. |
</threat_model>
